import os import json import uuid from datetime import datetime from fastapi import FastAPI, UploadFile, File, Form, HTTPException from fastapi.staticfiles import StaticFiles from fastapi.responses import HTMLResponse, FileResponse from pydantic import BaseModel from typing import Optional, List app = FastAPI(title="HR Management System") # إنشاء المجلدات تلقائياً os.makedirs("data", exist_ok=True) os.makedirs("uploads", exist_ok=True) # ملفات التخزين HIRES_FILE = "data/hires.json" RESIGNATIONS_FILE = "data/resignations.json" def load_json(filepath): if not os.path.exists(filepath): return [] with open(filepath, "r", encoding="utf-8") as f: return json.load(f) def save_json(filepath, data): with open(filepath, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) # Models class HireRecord(BaseModel): id: Optional[str] = None employee_name: str = "" branch: str = "" job_title: str = "" hiring_date: str = "" grading_system: str = "" net_basic_salary: float = 0 net_allowance: float = 0 net_kpi: float = 0 total_net: float = 0 educational_qualifications: str = "" age: int = 0 address: str = "" professional_experience: str = "" direct_manager: str = "" status: str = "Pending" manager_comment: str = "" photo_path: str = "" checklist: List[str] = [] created_at: str = "" class ResignationRecord(BaseModel): id: Optional[str] = None employee_name: str = "" branch: str = "" job_title: str = "" hiring_date: str = "" resignation_date: str = "" last_working_day: str = "" grading_system: str = "" total_net: float = 0 reason_for_leaving: str = "" notice_period_status: str = "" handover_status: str = "" total_experience: str = "" direct_manager: str = "" absence_days: int = 0 loan_balance: float = 0 status: str = "Pending" manager_comment: str = "" photo_path: str = "" created_at: str = "" # Routes @app.get("/", response_class=HTMLResponse) async def serve_frontend(): return FileResponse("index.html") @app.post("/api/upload-photo") async def upload_photo(file: UploadFile = File(...)): file_id = str(uuid.uuid4()) ext = file.filename.split(".")[-1] filename = f"{file_id}.{ext}" filepath = os.path.join("uploads", filename) with open(filepath, "wb") as f: content = await file.read() f.write(content) return {"photo_path": f"/uploads/{filename}"} @app.get("/uploads/{filename}") async def get_upload(filename: str): return FileResponse(os.path.join("uploads", filename)) # Hires CRUD @app.post("/api/hires") async def create_hire(record: HireRecord): records = load_json(HIRES_FILE) record.id = str(uuid.uuid4()) record.created_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S") records.append(record.dict()) save_json(HIRES_FILE, records) return {"success": True, "id": record.id} @app.get("/api/hires") async def get_hires(): return load_json(HIRES_FILE) @app.get("/api/hires/{record_id}") async def get_hire(record_id: str): records = load_json(HIRES_FILE) for r in records: if r["id"] == record_id: return r raise HTTPException(status_code=404, detail="Record not found") @app.put("/api/hires/{record_id}") async def update_hire(record_id: str, record: HireRecord): records = load_json(HIRES_FILE) for i, r in enumerate(records): if r["id"] == record_id: records[i] = record.dict() records[i]["id"] = record_id save_json(HIRES_FILE, records) return {"success": True} raise HTTPException(status_code=404, detail="Record not found") @app.delete("/api/hires/{record_id}") async def delete_hire(record_id: str): records = load_json(HIRES_FILE) records = [r for r in records if r["id"] != record_id] save_json(HIRES_FILE, records) return {"success": True} #
